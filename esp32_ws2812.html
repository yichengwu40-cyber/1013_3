對應的 Arduino 程式碼

完整支援序列埠指令解析（`ON:x`, `OFF:x`, `LED:x,r,g,b`）與感測器讀取。

#include <Adafruit_NeoPixel.h>

const int LDR_PIN = 39;     // 光感測器輸入 (ADC1_CH3)
const int POT_PIN = 34;     // 可變電阻輸入 (ADC1_CH6)
const int LED_PIN = 26;     // WS2812 資料引腳 (GPIO26)
const int NUM_PIXELS = 3;   // 3顆 WS2812 燈珠

Adafruit_NeoPixel pixels(NUM_PIXELS, LED_PIN, NEO_GRB + NEO_KHZ800);

bool ledState[NUM_PIXELS] = {true, true, true};
int ledRed[NUM_PIXELS] = {0, 255, 0};
int ledGreen[NUM_PIXELS] = {255, 0, 0};
int ledBlue[NUM_PIXELS] = {0, 0, 255};

void setup() {
  Serial.begin(115200);
  pixels.begin();
  pixels.show();
}

void loop() {
  if (Serial.available() > 0) {
    String command = Serial.readStringUntil('\n');
    command.trim();

    if (command.startsWith("ON:") || command.startsWith("OFF:")) {
      bool state = command.startsWith("ON:");
      int index = command.substring(3).toInt();
      if (index >= 0 && index < NUM_PIXELS) {
        ledState[index] = state;
      }
    }
    else if (command.startsWith("LED:")) {
      int firstComma = command.indexOf(',');
      int secondComma = command.indexOf(',', firstComma + 1);
      int thirdComma = command.indexOf(',', secondComma + 1);
      
      if (firstComma > 0 && secondComma > 0 && thirdComma > 0) {
        int index = command.substring(4, firstComma).toInt();
        if (index >= 0 && index < NUM_PIXELS) {
          ledRed[index] = command.substring(firstComma + 1, secondComma).toInt();
          ledGreen[index] = command.substring(secondComma + 1, thirdComma).toInt();
          ledBlue[index] = command.substring(thirdComma + 1).toInt();
          ledState[index] = true;
        }
      }
    }
  }

  int ldrValue = analogRead(LDR_PIN);
  int potValue = analogRead(POT_PIN);
  float brightnessFactor = (float)potValue / 4095.0;

  for (int i = 0; i < NUM_PIXELS; i++) {
    if (!ledState[i]) {
      pixels.setPixelColor(i, pixels.Color(0, 0, 0));
    } else {
      int r = ledRed[i] * brightnessFactor;
      int g = ledGreen[i] * brightnessFactor;
      int b = ledBlue[i] * brightnessFactor;
      pixels.setPixelColor(i, pixels.Color(r, g, b));
    }
  }
  pixels.show();

  Serial.print("LDR: ");
  Serial.print(ldrValue);
  Serial.print(" | Pot: ");
  Serial.println(potValue);

  delay(100);
}
